-- Crear la base de datos
CREATE DATABASE IF NOT EXISTS deepsky_db;
USE deepsky_db;

-- ==========================================
-- TABLAS SIN DEPENDENCIAS (Catálogos)
-- ==========================================

CREATE TABLE USUARIO (
    ID_USUARIO INT AUTO_INCREMENT PRIMARY KEY,
    NOMBRE VARCHAR(80) NOT NULL,
    EMAIL VARCHAR(120) NOT NULL UNIQUE,
    PASSWORD_HASH VARCHAR(255) NOT NULL,
    ROL ENUM('PARTICIPANTE', 'DIVULGADOR') NOT NULL DEFAULT 'PARTICIPANTE',
    FECHA_REGISTRO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE CATEGORIA (
    ID_CATEGORIA INT AUTO_INCREMENT PRIMARY KEY,
    NOMBRE VARCHAR(80) NOT NULL UNIQUE,
    DESCRIPCION TEXT
);

CREATE TABLE IMAGEN_NASA (
    ID_IMAGEN INT AUTO_INCREMENT PRIMARY KEY,
    NASA_ID VARCHAR(100) NOT NULL UNIQUE COMMENT 'ID original de la API NASA Images',
    URL_IMAGEN VARCHAR(500) NOT NULL,
    TITULO VARCHAR(255),
    DESCRIPCION TEXT,
    FECHA_CREACION DATE,
    KEYWORDS TEXT COMMENT 'Palabras clave separadas por coma',
    FECHA_CACHE DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IMAGEN_DIA (
    ID_IMAGEN_DIA INT AUTO_INCREMENT PRIMARY KEY,
    URL_IMAGEN VARCHAR(500) NOT NULL,
    TITULO VARCHAR(255),
    DESCRIPCION TEXT
);

CREATE TABLE PLANETA (
    ID_PLANETA INT AUTO_INCREMENT PRIMARY KEY,
    NOMBRE VARCHAR(80) NOT NULL UNIQUE,
    MASA_KG DECIMAL(30,4),
    RADIO_KM DECIMAL(14,4),
    PERIODO_ORBITAL_DIAS DECIMAL(14,4),
    DISTANCIA_SOL_UA DECIMAL(14,8),
    INCLINACION_AXIAL DECIMAL(8,4),
    TEXTURA_URL VARCHAR(500) COMMENT 'Asset de textura para Three.js / WebGL',
    MODELO_3D_URL VARCHAR(500) COMMENT 'Asset glTF/OBJ',
    DESCRIPCION TEXT,
    ORDEN_VISUAL INT DEFAULT 0
);

-- ==========================================
-- TABLAS CON DEPENDENCIAS
-- ==========================================

CREATE TABLE PREFERENCIA_NOTIFICACION (
    ID_PREFERENCIA INT AUTO_INCREMENT PRIMARY KEY,
    USUARIO_ID INT NOT NULL UNIQUE,
    RECIBIR_EVENTOS BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO)
);

CREATE TABLE FAVORITO_IMAGEN (
    ID_FAVORITO INT AUTO_INCREMENT PRIMARY KEY,
    USUARIO_ID INT NOT NULL,
    IMAGEN_ID INT NOT NULL,
    FECHA_GUARDADO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (IMAGEN_ID) REFERENCES IMAGEN_NASA(ID_IMAGEN),
    UNIQUE INDEX uq_favorito_imagen (USUARIO_ID, IMAGEN_ID)
);

CREATE TABLE FORO (
    ID_FORO INT AUTO_INCREMENT PRIMARY KEY,
    CATEGORIA_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL COMMENT 'Backend valida que el rol sea DIVULGADOR',
    TITULO VARCHAR(255) NOT NULL,
    CONTENIDO TEXT NOT NULL,
    FECHA_CREACION DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (CATEGORIA_ID) REFERENCES CATEGORIA(ID_CATEGORIA),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO)
);

CREATE TABLE COMENTARIO_FORO (
    ID_COMENTARIO INT AUTO_INCREMENT PRIMARY KEY,
    FORO_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL,
    PADRE_ID INT DEFAULT NULL COMMENT 'NULL = raíz | valor = respuesta anidada',
    CONTENIDO TEXT NOT NULL,
    FECHA_COMENTARIO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (FORO_ID) REFERENCES FORO(ID_FORO),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (PADRE_ID) REFERENCES COMENTARIO_FORO(ID_COMENTARIO)
);

CREATE TABLE ARCHIVO_ADJUNTO (
    ID_ARCHIVO INT AUTO_INCREMENT PRIMARY KEY,
    COMENTARIO_ID INT NOT NULL,
    URL_IMAGEN VARCHAR(500) NOT NULL,
    NOMBRE_ARCHIVO VARCHAR(200),
    FOREIGN KEY (COMENTARIO_ID) REFERENCES COMENTARIO_FORO(ID_COMENTARIO)
);

CREATE TABLE EVENTO_ASTRONOMICO (
    ID_EVENTO INT AUTO_INCREMENT PRIMARY KEY,
    USUARIO_ID INT NOT NULL COMMENT 'Backend valida que el rol sea DIVULGADOR',
    TITULO VARCHAR(200) NOT NULL,
    DESCRIPCION TEXT,
    FECHA_HORA DATETIME NOT NULL,
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO)
);

CREATE TABLE FAVORITO_EVENTO (
    ID_FAVORITO_EVENTO INT AUTO_INCREMENT PRIMARY KEY,
    USUARIO_ID INT NOT NULL,
    EVENTO_ID INT NOT NULL,
    FECHA_GUARDADO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (EVENTO_ID) REFERENCES EVENTO_ASTRONOMICO(ID_EVENTO),
    UNIQUE INDEX uq_favorito_evento (USUARIO_ID, EVENTO_ID)
);

CREATE TABLE COMENTARIO_IMAGEN_DIA (
    ID_COMENTARIO INT AUTO_INCREMENT PRIMARY KEY,
    IMAGEN_DIA_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL,
    PADRE_ID INT DEFAULT NULL COMMENT 'Soporte para respuestas anidadas',
    CONTENIDO TEXT NOT NULL,
    FECHA_COMENTARIO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (IMAGEN_DIA_ID) REFERENCES IMAGEN_DIA(ID_IMAGEN_DIA),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (PADRE_ID) REFERENCES COMENTARIO_IMAGEN_DIA(ID_COMENTARIO)
);

CREATE TABLE RETO_FOTOGRAFIA (
    ID_RETO INT AUTO_INCREMENT PRIMARY KEY,
    USUARIO_ID INT NOT NULL COMMENT 'Backend valida que el rol sea DIVULGADOR',
    CATEGORIA_ID INT NOT NULL,
    TEMA VARCHAR(200) NOT NULL,
    DESCRIPCION TEXT,
    REGLAS TEXT,
    URL_IMAGEN_REFERENCIA VARCHAR(500) NOT NULL,
    FECHA_INICIO DATETIME NOT NULL,
    FECHA_FIN DATETIME NOT NULL,
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (CATEGORIA_ID) REFERENCES CATEGORIA(ID_CATEGORIA)
);

CREATE TABLE PARTICIPACION_RETO (
    ID_PARTICIPACION INT AUTO_INCREMENT PRIMARY KEY,
    RETO_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL,
    URL_FOTOGRAFIA VARCHAR(500) NOT NULL,
    TITULO VARCHAR(200),
    DESCRIPCION TEXT,
    FECHA_SUBIDA DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    APROBADA BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (RETO_ID) REFERENCES RETO_FOTOGRAFIA(ID_RETO),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    UNIQUE INDEX uq_una_participacion_por_reto (RETO_ID, USUARIO_ID)
);

CREATE TABLE VOTO_RETO (
    ID_VOTO INT AUTO_INCREMENT PRIMARY KEY,
    PARTICIPACION_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL,
    PUNTUACION TINYINT NOT NULL COMMENT 'Valor del 1 al 5',
    FECHA_VOTO DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (PARTICIPACION_ID) REFERENCES PARTICIPACION_RETO(ID_PARTICIPACION),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO),
    UNIQUE INDEX uq_un_voto_por_participacion (PARTICIPACION_ID, USUARIO_ID)
);

CREATE TABLE DATO_CURIOSO_PLANETA (
    ID_DATO INT AUTO_INCREMENT PRIMARY KEY,
    PLANETA_ID INT NOT NULL,
    USUARIO_ID INT NOT NULL COMMENT 'Backend valida que el rol sea DIVULGADOR',
    DATO TEXT NOT NULL,
    FECHA_PUBLICACION DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (PLANETA_ID) REFERENCES PLANETA(ID_PLANETA),
    FOREIGN KEY (USUARIO_ID) REFERENCES USUARIO(ID_USUARIO)
);
